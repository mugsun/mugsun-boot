package com.mugsun.boot.demo.controller;

import cn.dev33.satoken.annotation.SaCheckLogin;
import cn.dev33.satoken.annotation.SaCheckPermission;
import com.mugsun.boot.demo.entity.Category;
import com.mugsun.boot.demo.mapper.CategoryMapper;
import com.mugsun.core.tool.api.R;
import com.mybatisflex.core.paginate.Page;
import com.mybatisflex.core.query.QueryWrapper;
import org.springframework.web.bind.annotation.*;

import java.util.List;

/**
 * 分类管理
 *
 * @author mugsun
 */
@RestController
@RequestMapping("/demo/category")
@SaCheckLogin
public class CategoryController {

	private final CategoryMapper categoryMapper;

	public CategoryController(CategoryMapper categoryMapper) {
		this.categoryMapper = categoryMapper;
	}

	/** 懒加载树：parentId 空取根节点（父列=0），否则取其直接子节点；分组统计 hasChildren 驱动前端展开箭头（防大树全量加载） */
	@GetMapping("/tree")
	public R<List<Category>> tree(@RequestParam(required = false) Long parentId) {
		List<Category> list = categoryMapper.selectListByQuery(QueryWrapper.create()
			.eq("parent_id", parentId == null ? 0L : parentId).orderBy("id", true));
		if (!list.isEmpty()) {
			java.util.Map<Long, Long> childCounts = new java.util.HashMap<>();
			for (com.mybatisflex.core.row.Row row : com.mybatisflex.core.row.Db.selectListBySql(
				"SELECT parent_id AS pid, COUNT(*) AS cnt FROM gen_category WHERE is_deleted = 0 GROUP BY parent_id")) {
				childCounts.put(row.getLong("pid"), row.getLong("cnt"));
			}
			list.forEach(n -> n.setHasChildren(childCounts.getOrDefault(n.getId(), 0L) > 0));
		}
		return R.data(list);
	}

	@GetMapping("/page")
	public R<Page<Category>> page(@RequestParam(defaultValue = "1") long pageNum,
			@RequestParam(defaultValue = "10") long pageSize) {
		return R.data(categoryMapper.paginate(pageNum, pageSize, QueryWrapper.create().orderBy("id", false)));
	}

	@GetMapping("/detail")
	public R<Category> detail(@RequestParam Long id) {
		return R.data(categoryMapper.selectOneById(id));
	}

	@SaCheckPermission("demo:category:save")
	@PostMapping("/submit")
	public R<Void> submit(@RequestBody Category category) {
		if (category.getId() == null) {
			categoryMapper.insertSelective(category);
		} else {
			categoryMapper.update(category);
		}
		return R.success("操作成功");
	}

	@SaCheckPermission("demo:category:remove")
	@PostMapping("/remove")
	public R<Void> remove(@RequestBody List<Long> ids) {
		categoryMapper.deleteBatchByIds(ids);
		return R.success("删除成功");
	}
}
